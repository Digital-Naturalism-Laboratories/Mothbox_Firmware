# ===========================================================================
# Mothbox Pro reference /boot/firmware/config.txt        (Raspberry Pi 5)
#
# Known-good copy taken from a working Mothbox Pro, September 2026. Every line
# the Mothbox needs is in here. Keep it: a single corrupted FAT sector wiped
# the top of a live config.txt once (it took i2c and spi with it), and a fresh
# image starts without any of the [all] section below.
#
# To restore a box from this file:
#   sudo cp /boot/firmware/config.txt /boot/firmware/config.txt.bak
#   sudo cp /boot/firmware/mothbox_custom/reference_boot_config.txt /boot/firmware/config.txt
#   sudo reboot
# (or copy it over from this repo if mothbox_custom hasn't been deployed yet)
#
# Then confirm all four hardware interfaces came back:
#   ls /dev/i2c-1 /dev/spidev0.0 /sys/bus/w1/devices/ && rpicam-hello --list-cameras | head -3
#
# Lines the Mothbox depends on -- do not remove:
#   dtparam=i2c_arm=on                                voltage sensor, config switches, light sensor
#   dtparam=spi=on  +  dtoverlay=spi0-1cs,no_miso      e-paper display; no_miso hands GPIO 9 (MISO)
#                                                      back to attract relay Ch3
#   dtoverlay=ov64a40,cam0,link-frequency=456000000    Arducam 64MP OWLSIGHT (OV64A40). This is NOT
#                                                      the arducam-64mp / Hawkeye overlay.
#                                                      456 MHz = Arducam 'high speed' mode: full-res
#                                                      readout 2.60 fps vs 2.02 at 360 MHz, so ~30%
#                                                      less rolling-shutter smear on moving insects.
#                                                      Drop to 360000000 if photos show tearing.
#   dtparam=rtc_bbat_vchg=3000000                      trickle-charges the RTC coin cell so wake
#                                                      alarms survive a power loss
#   dtoverlay=w1-gpio,gpiopin=4                        DS18B20 board temperature (Diagnostics.py)
#   PSU_MAX_CURRENT=5000  +  usb_max_current_enable=1  full USB current for the backup drive
#   dtoverlay=disable-bt                               Bluetooth off (power)
#
# Everything below this block is the file exactly as it runs on the box.
# ===========================================================================

# For more options and information see
# http://rptl.io/configtxt
# Some settings may impact device functionality. See link above for details

# Uncomment some or all of these to enable the optional hardware interfaces
dtparam=i2c_arm=on
#dtparam=i2s=on
dtparam=spi=on
dtoverlay=spi0-1cs,no_miso

# Enable audio (loads snd_bcm2835)
dtparam=audio=on

# Additional overlays and parameters are documented
# /boot/firmware/overlays/README

# Automatically load overlays for detected cameras
camera_auto_detect=1

# Automatically load overlays for detected DSI displays
display_auto_detect=1

# Automatically load initramfs files, if found
auto_initramfs=1

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d, cma-512
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

[cm4]
# Enable host mode on the 2711 built-in XHCI USB controller.
# This line should be removed if the legacy DWC2 controller is required
# (e.g. for USB device mode) or if USB support is not required.
otg_mode=1

[all]
dtoverlay=ov64a40,cam0,link-frequency=456000000
dtparam=rtc_bbat_vchg=3000000
dtoverlay=disable-bt

PSU_MAX_CURRENT=5000
usb_max_current_enable=1

dtoverlay=w1-gpio,gpiopin=4
